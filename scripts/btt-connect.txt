NVIM_CONFIG_DIR="$HOME/.config/nvim"

notify() {
    local message="$1"
    /usr/bin/osascript -e "display notification \"$message\" with title \"Neovim Config\"" 2>/dev/null
}

cd "$NVIM_CONFIG_DIR" || exit 1
current_branch=$(/usr/bin/git branch --show-current 2>/dev/null)

if [[ "$current_branch" != "glove80" ]]; then
    if ! /usr/bin/git diff --quiet 2>/dev/null || ! /usr/bin/git diff --cached --quiet 2>/dev/null; then
        /usr/bin/git stash push -m "Auto-stash for Glove80 - $(date)" 2>/dev/null
    fi
    
    if /usr/bin/git checkout glove80 2>/dev/null; then
        notify "🎯 Switched to Glove80 configuration"
        echo "$(date): Switched to glove80 branch" >> "$NVIM_CONFIG_DIR/.switch-log"
    else
        notify "❌ Failed to switch to Glove80 configuration"
    fi
else
    notify "🎯 Already using Glove80 configuration"
fi
